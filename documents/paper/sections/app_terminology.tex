\section{Terminology}

\begin{itemize}
    \item Benchmark \(b\): HellaSwag.
    \item Language task \(b_\ell\): HellaSwag-es, HellaSwag-en, HellaSwag-fr.
    \item Sample/example: one question or instance inside HellaSwag-es.
    \item Evaluation suite: the complete collection of benchmark families and tasks.
\end{itemize}

    \begin{tabular}{lll}

        \textbf{Model configuration} & $\theta$: \((N,A,K,M,T)\) & A seed replicate \\
        \textbf{Training run} & One configuration trained with seed \(S\) & One task evaluation \\
        \textbf{Checkpoint} & Saved model state at fraction \(f\) & The fraction itself \\
        \textbf{Training fraction} & \(f\in\{0.1,\ldots,1.0,0.85,0.95\}\) & A model artifact \\
        \textbf{Model variant} & One level of an intervention being ranked & A proxy/reference checkpoint \\
        \textbf{Proxy point} & Size and training fraction used for the inexpensive ranking & The model variant \\
        \textbf{Reference point} & Size and fraction defining the target ranking & Necessarily 1.7B \\
        \textbf{Benchmark} & HellaSwag & HellaSwag-es \\
        \textbf{Language task} & HellaSwag-es & The complete HellaSwag family \\
        \textbf{Sample/example} & One question & A task \\
        \textbf{Score observation} & One checkpoint evaluated on one task & ``Evaluation run'' \\
        \textbf{Ranking agreement} & General concept & A numerical metric \\
        \textbf{Decision accuracy} & Pairwise ranking-agreement metric & General reliability \\
        \textbf{Scaling predictability} & \(R^2\) of a scaling fit & Decision accuracy \\
        \textbf{Scaling monotonicity} & Spearman correlation along a scaling curve & Rank transfer \\
        \textbf{Reliability} & Broad interpretive concept & A metric unless explicitly defined \\
        \bottomrule
    \end{tabular}
